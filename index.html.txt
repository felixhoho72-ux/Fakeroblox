const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
function playSound(type) {
  try {
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain); gain.connect(audioCtx.destination);
    const now = audioCtx.currentTime;
    if (type === 'jump') {
      osc.frequency.setValueAtTime(280, now);
      osc.frequency.exponentialRampToValueAtTime(620, now + 0.12);
      gain.gain.setValueAtTime(0.14, now);
      gain.gain.exponentialRampToValueAtTime(0.01, now + 0.15);
      osc.start(now); osc.stop(now + 0.15);
    } else if (type === 'collect') {
      osc.frequency.setValueAtTime(780, now);
      osc.frequency.exponentialRampToValueAtTime(1300, now + 0.09);
      gain.gain.setValueAtTime(0.11, now);
      gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
      osc.start(now); osc.stop(now + 0.12);
    } else if (type === 'buy' || type === 'code') {
      osc.type = 'square';
      osc.frequency.setValueAtTime(420, now);
      osc.frequency.setValueAtTime(680, now + 0.1);
      gain.gain.setValueAtTime(0.1, now);
      gain.gain.exponentialRampToValueAtTime(0.01, now + 0.22);
      osc.start(now); osc.stop(now + 0.22);
    }
  } catch(e){}
}

function showCategory(cat, el) {
  document.querySelectorAll('.shop-tab').forEach(t => t.classList.remove('active'));
  if (el) el.classList.add('active');
  const grid = document.getElementById('items-grid');
  grid.innerHTML = '';
  catalog[cat].forEach(item => {
    const ownedItem = owned.has(item.id);
    const card = document.createElement('div');
    card.className = 'item-card';
    card.innerHTML = `<div class="item-icon">${item.icon}</div><div class="item-name">${item.name}</div><div class="item-price">${ownedItem ? 'Owned' : 'R$ ' + item.price}</div><button class="buy-btn" onclick="buyItem('${cat}','${item.id}')">${ownedItem ? 'Equip' : 'Buy'}</button>`;
    grid.appendChild(card);
  });
}

function buyItem(cat, id) {
  const item = catalog[cat].find(i => i.id === id);
  if (!item) return;
  if (!owned.has(id)) {
    if (robux < item.price) { alert('Not enough Robux!'); return; }
    robux -= item.price;
    owned.add(id);
    updateRobux();
    playSound('buy');
  }
  if (cat === 'hats') {
    equipped.hat = item.icon;
    document.getElementById('eq-hat').textContent = item.icon;
    document.getElementById('p-hat').textContent = item.icon;
  } else if (cat === 'faces') {
    equipped.face = item.icon;
    document.getElementById('eq-face').textContent = item.icon;
  } else if (cat === 'clothes') {
    equipped.clothes = item.color;
    document.getElementById('big-body').style.background = item.color;
    document.getElementById('p-body').style.background = item.color;
  }
  showCategory(cat);
}

function redeemCode() {
  const val = document.getElementById('promo-input').value.trim();
  if (val === '9211') {
    robux = Infinity;
    updateRobux();
    playSound('code');
    alert('🎉 Infinite Robux Unlocked!');
    document.getElementById('promo-input').value = '';
    return;
  }
  const amount = parseInt(val);
  if (!isNaN(amount) && amount > 0) {
    robux += amount;
    updateRobux();
    playSound('buy');
    alert(`+${amount} Robux added!`);
    document.getElementById('promo-input').value = '';
    return;
  }
  alert('Invalid code');
}

function checkCollect() {
  document.querySelectorAll('[data-collect]').forEach(el => {
    const r = el.getBoundingClientRect();
    const pr = document.getElementById('player').getBoundingClientRect();
    if (Math.abs(r.left - pr.left) < 48 && Math.abs(r.top - pr.top) < 58) {
      el.remove();
      score += el.classList.contains('fruit') ? 50 : el.classList.contains('pet') ? 30 : 10;
      document.getElementById('score').textContent = 'Score: ' + score;
      playSound('collect');
    }
  });
}

const joy = document.getElementById('joystick');

let active = false, sx, sy;
function start(e){active=true;const t=e.touches?e.touches[0]:e;const r=joy.getBoundingClientRect();sx=r.left+r.width/2;sy=r.top+r.height/2;move(t.clientX,t.clientY)}
function move(x,y){if(!active)return;let dx=x-sx,dy=y-sy;const max=50,d=Math.hypot(dx,dy);if(d>max){dx*=max/d;dy*=max/d}knob.style.transform=`translate(calc(-50% + ${dx}px),calc(-50% + ${dy}px))`;vx=(dx/max)*MAX_SPEED;vy=(dy/max)*MAX_SPEED}
function end(){active=false;knob.style.transform='translate(-50%,-50%)';vx=0;vy=0}

joy.addEventListener('touchend',end);
joy.addEventListener('mousedown',start);

function updateCamera(){
  const camX=-(playerX-window.innerWidth/2);
  const camY=-(playerY-window.innerHeight/2);
  document.getElementById('world').style.transform=`translate(${camX}px,${camY}px)`;
  document.getElementById('player').style.left=playerX+'px';
  document.getElementById('player').style.top=playerY+'px';
}

function loop(){
  if(document.getElementById('game-screen').style.display==='block'){
    playerX+=vx;playerY+=vy;
    playerX=Math.max(60,Math.min(WORLD_SIZE-60,playerX));
    playerY=Math.max(60,Math.min(WORLD_SIZE-60,playerY));
    updateCamera();
    checkCollect();
  }
  requestAnimationFrame(loop);
}
